// AWS Lambda function — Bedrock Claude Haiku 4.5 proxy (ESM version) import https from 'https'; const BEDROCK_API_KEY = process.env.BEDROCK_API_KEY; const BEDROCK_HOST = 'bedrock-mantle.us-east-1.api.aws'; const MODEL_ID = 'anthropic.claude-haiku-4-5'; const SYSTEM_PROMPT = `You are a friendly, professional AWS pre-meeting discovery assistant. Your job is to have a natural conversation to understand what the customer cares about before their first meeting with an AWS representative. You are embedded in a guided chatbot that asks structured questions. The user may type free-text answers instead of clicking pre-defined options. Your role is to: 1. Acknowledge their free-text response naturally (1-2 sentences max) 2. Extract what's relevant from their answer 3. Gently guide them back if they went off-topic CONTEXT: You will receive the full conversation history including: - The structured questions that were asked - The user's previous answers (both pill-selections and free-text) - The current question being asked TONE & PERSONALITY: - Warm but efficient — like a great account manager, not a chatbot - Concise — never repeat what the customer already said - Natural — use contractions, vary your sentence structure, never sound formulaic - Customer-obsessed — every response shows you care about THEIR outcome ANTI-ROBOTIC RULES: - NEVER start two consecutive responses with "Thank you" or "Thanks for sharing" - NEVER use the same transition phrase twice in a conversation - NEVER say "That's really helpful!" more than once - Vary acknowledgements: brief reactions tied to what they actually said - Sometimes skip the acknowledgement entirely and go straight to the next question - Use natural bridges like: "Got it —", "Makes sense.", "Interesting —", "So [reflect what they said]." RESPONSE FORMAT: Always respond with valid JSON only — no markdown, no code blocks, no explanation: {"relevant": true, "reply": "your acknowledgement here", "extracted": "the key info you extracted from their answer"} The "extracted" field should contain a concise summary of what their answer means in the context of the question (e.g., if the question was about AWS services and they said "we use S3 and some EC2 instances", extracted should be "storage, compute"). RELEVANCE CLASSIFICATION: Set "relevant": true if the response relates (even loosely) to: - Their business, industry, role, or company - A challenge, goal, or project - Technology, cloud, or digital transformation - Timeline, budget, or decision-making - Any reasonable answer to the current question Set "relevant": false if: - Clearly unrelated to any professional context (personal emergencies, jokes, random text, testing the system) - Contains only greetings with no substance WHEN "relevant": false (off-topic message): Respond with empathy + gentle redirect. The customer is a human — acknowledge briefly, then guide them back. Never ignore what they said. Response pattern: 1. Brief, genuine acknowledgement (1 short sentence max) 2. Gentle redirect back to the question or offer to skip Examples: - "my house is on fire" -> {"relevant": false, "reply": "I hope everything is okay! Feel free to come back anytime, or skip ahead if you'd like.", "extracted": ""} - "asdfghjkl" -> {"relevant": false, "reply": "No problem — if you'd like, you can pick one of the options above or skip this one.", "extracted": ""} - "tell me a joke" -> {"relevant": false, "reply": "Ha — I wish I were funnier! I'm better at helping prep your meeting though. Want to pick an option above, or skip this one?", "extracted": ""} KEY RULES: - ALWAYS acknowledge the human moment first (even 3 words is enough) - Never sound dismissive or transactional - Keep it to 1-2 sentences max for the reply - The "extracted" field helps the system map their free-text to structured categories - If you can map their answer to one of the available options for the current question, mention it in "extracted" ERROR HANDLING: If you cannot generate a valid response for any reason, return: {"relevant": false, "reply": "Something went wrong on my end — sorry about that! You can try again, pick one of the options above, or skip this question.", "extracted": ""}`; export const handler = async (event) => { if (event.requestContext?.http?.method === 'OPTIONS') { return { statusCode: 200, headers: cors(), body: '' }; } try { const body = JSON.parse(event.body || '{}'); const userMessage = body.message; const conversationHistory = body.history || []; const currentQuestion = body.currentQuestion || ''; const availableOptions = body.availableOptions || []; if (!userMessage || typeof userMessage !== 'string' || userMessage.trim().length === 0) { return resp(400, { error: 'Missing or empty message field.' }); } if (userMessage.length > 2000) { return resp(400, { error: 'Message too long.' }); } // Handle resource recommendation request if (currentQuestion === 'RESOURCE_RECOMMENDATION') { return await handleResourceRecommendation(conversationHistory); } // Build messages array with conversation context const messages = []; // Add conversation history as context if (conversationHistory.length > 0) { let contextSummary = "Here's the conversation so far:\n"; conversationHistory.forEach((entry, i) => { if (entry.role === 'bot') { contextSummary += `Bot asked: ${entry.text}\n`; } else if (entry.role === 'user') { contextSummary += `Customer answered: ${entry.text}\n`; } }); messages.push({ role: 'user', content: contextSummary }); messages.push({ role: 'assistant', content: 'Got it, I have the context from the conversation so far.' }); } // Add current question context let currentContext = ''; if (currentQuestion) { currentContext += `The current question being asked is: "${currentQuestion}"\n`; } if (availableOptions.length > 0) { currentContext += `The pre-defined options available are: ${availableOptions.join(', ')}\n`; } currentContext += `\nThe customer just typed: "${userMessage.trim()}"\n\nPlease acknowledge their response and extract the relevant info.`; messages.push({ role: 'user', content: currentContext }); const payload = JSON.stringify({ model: MODEL_ID, max_tokens: 200, system: SYSTEM_PROMPT, messages: messages }); const result = await callBedrock(payload); const parsed = JSON.parse(result); let rawText = parsed.content?.[0]?.text || '{"relevant":true,"reply":"Thanks for sharing that.","extracted":""}'; // Strip markdown code blocks if Claude wraps the JSON rawText = rawText.replace(/```json\s*/g, '').replace(/```\s*/g, '').trim(); // Parse Claude's JSON response let aiResponse; try { aiResponse = JSON.parse(rawText); } catch(e) { aiResponse = { relevant: true, reply: rawText, extracted: '' }; } return resp(200, { reply: aiResponse.reply || 'Thanks for sharing that.', relevant: aiResponse.relevant !== false, extracted: aiResponse.extracted || '' }); } catch (err) { console.error('Error:', err); return resp(200, { reply: 'Something went wrong on my end — sorry about that! You can try again, pick one of the options above, or skip this question.', relevant: false, extracted: '' }); } }; async function handleResourceRecommendation(conversationHistory) { const RESOURCE_PROMPT = `You are an AWS pre-meeting assistant. Based on the customer's conversation below, recommend exactly 3 specific AWS resources (articles, guides, case studies, or tools) that are most relevant to what they discussed. RULES: - Pick resources specific to their industry, use case, and concerns - Each resource should have a title and a brief one-line description - Use real AWS resource names (e.g., "Amazon Bedrock Getting Started Guide", "AWS Cost Explorer", "AWS Well-Architected Framework") - Match the resource to what they actually said, not generic recommendations RESPONSE FORMAT (JSON only, no markdown): {"resources": [ {"icon": "emoji", "text": "Resource title - brief description"}, {"icon": "emoji", "text": "Resource title - brief description"}, {"icon": "emoji", "text": "Resource title - brief description"} ]} Use relevant emojis as HTML entities: 📄 (document), 🤖 (robot/AI), 💰 (money), 🔒 (security), 📐 (chart), ☁ (cloud), 🎥 (video)`; let contextSummary = "Here's the full conversation with the customer:\n"; conversationHistory.forEach((entry) => { if (entry.role === 'bot') { contextSummary += `Bot: ${entry.text}\n`; } else if (entry.role === 'user') { contextSummary += `Customer: ${entry.text}\n`; } }); contextSummary += "\nBased on this conversation, recommend 3 tailored resources."; const payload = JSON.stringify({ model: MODEL_ID, max_tokens: 300, system: RESOURCE_PROMPT, messages: [{ role: 'user', content: contextSummary }] }); try { const result = await callBedrock(payload); const parsed = JSON.parse(result); let rawText = parsed.content?.[0]?.text || ''; rawText = rawText.replace(/```json\s*/g, '').replace(/```\s*/g, '').trim(); const aiResponse = JSON.parse(rawText); return resp(200, { reply: '', relevant: true, resources: aiResponse.resources || [] }); } catch (err) { console.error('Resource recommendation error:', err); return resp(200, { reply: '', relevant: true, resources: [] }); } } function callBedrock(payload) { return new Promise((resolve, reject) => { const options = { hostname: BEDROCK_HOST, port: 443, path: '/anthropic/v1/messages', method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': BEDROCK_API_KEY, 'anthropic-version': '2023-06-01', 'Content-Length': Buffer.byteLength(payload) } }; const req = https.request(options, (res) => { let data = ''; res.on('data', chunk => { data += chunk; }); res.on('end', () => { if (res.statusCode >= 200 && res.statusCode < 300) { resolve(data); } else { reject(new Error(`Bedrock ${res.statusCode}: ${data}`)); } }); }); req.on('error', reject); req.setTimeout(15000, () => { req.destroy(); reject(new Error('Timeout')); }); req.write(payload); req.end(); }); } function cors() { return { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'Content-Type', 'Access-Control-Allow-Methods': 'POST, OPTIONS' }; } function resp(statusCode, body) { return { statusCode, headers: cors(), body: JSON.stringify(body) }; }
